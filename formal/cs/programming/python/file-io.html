Reading files: basics


          # read entire file
          file = open("data.txt", "r")
          content = file.read()
          file.close()
          # read with context manager (recommended)
          with open("data.txt", "r") as file:
              content = file.read()

Reading files: line by line


          # read line by line
          with open("data.txt", "r") as file:
              for line in file:
                  print(line.strip())
          # read all lines into list
          with open("data.txt", "r") as file:
              lines = file.readlines()

Reading methods: read and readline


          # read() - read entire file or n characters
          with open("data.txt", "r") as file:
              content = file.read()        # all content
              content = file.read(10)      # first 10 characters
          # readline() - read one line
          with open("data.txt", "r") as file:
              line1 = file.readline()
              line2 = file.readline()

Watch the cursor: right after read() it is at the end, so the read(10) on the next line returns ''. On a freshly opened file, read(10) gives the first 10 characters.

Reading methods: readlines and iteration


          # readlines() - read all lines
          with open("data.txt", "r") as file:
              lines = file.readlines()     # list of lines
          # iterate (memory efficient)
          with open("data.txt", "r") as file:
              for line in file:
                  print(line.strip())

Writing files


          # write (overwrites file)
          with open("output.txt", "w") as file:
              file.write("Hello, World!\n")
              file.write("Second line\n")
          # write multiple lines
          lines = ["Line 1\n", "Line 2\n", "Line 3\n"]
          with open("output.txt", "w") as file:
              file.writelines(lines)

Appending, and the missing newline


          # append (adds to existing file)
          with open("output.txt", "a") as file:
              file.write("Appended line\n")
          # write doesn't add newline automatically
          with open("output.txt", "w") as file:
              file.write("No newline here")
              file.write("This is on the same line")

          # read modes
          open("file.txt", "r")      # read (default)

          # append modes
          open("file.txt", "a")      # append

          # read and write
          open("file.txt", "r+")     # read and write
          open("file.txt", "w+")     # write and read (overwrites)
          open("file.txt", "a+")     # append and read
          # exclusive creation
          open("file.txt", "x")      # create (fails if exists)

File position


          with open("data.txt", "r") as file:
              # get current position
              pos = file.tell()
              # read some data
              data = file.read(10)
              # get new position
              pos = file.tell()
              # move to beginning
              file.seek(0)
              # move to specific position
              file.seek(10)
              # move relative to current position
              file.seek(5, 1)
              # move relative to end
              file.seek(-10, 2)

The last two calls need binary mode ("rb"): in text mode Python only allows seek(0, 1) and seek(0, 2) and raises io.UnsupportedOperation otherwise. The picture shows what they do in binary mode.

Binary files


          # write binary
          data = bytes([65, 66, 67, 68])  # ABCD
          with open("data.bin", "wb") as file:
              file.write(data)
          # read binary
          with open("data.bin", "rb") as file:
              data = file.read()
              print(data)                  # b'ABCD'

Copying and chunking binary files


        # copy binary file
        with open("image.png", "rb") as src:
            with open("copy.png", "wb") as dst:
                dst.write(src.read())
        # read binary in chunks
        with open("large.bin", "rb") as file:
            while True:
                chunk = file.read(1024)   # 1KB chunks
                if not chunk:
                    break
                process(chunk)

Chunks keep memory use small: only 1 KB is held at a time, and an empty chunk means the end of the file.

pathlib: reading, writing and parts


        # read and write
        text = path.read_text()
        path.write_text("Hello, World!")
        data = path.read_bytes()
        path.write_bytes(b"binary data")
        # path operations

        path.stem                          # "data"

Directory operations: list, create, remove


        import os
        # list directory
        files = os.listdir(".")
        files = os.listdir("/path/to/dir")
        # create directory
        os.mkdir("new_folder")
        os.makedirs("path/to/folder")      # create intermediate dirs
        # remove directory
        os.rmdir("folder")                 # only if empty
        import shutil
        shutil.rmtree("folder")            # remove with contents

shutil.rmtree deletes everything inside without asking. Double-check the path.

Working with CSV: reading


        import csv
        # read CSV
        with open("data.csv", "r") as file:
            reader = csv.reader(file)
            for row in reader:
                print(row)                 # row is a list
        # read CSV with headers
        with open("data.csv", "r") as file:
            reader = csv.DictReader(file)
            for row in reader:
                print(row["name"], row["age"])

Working with CSV: writing


        # write CSV
        data = [["Name", "Age"], ["Alice", 25], ["Bob", 30]]
        with open("output.csv", "w", newline="") as file:
            writer = csv.writer(file)
            writer.writerows(data)
        # write CSV with DictWriter
        with open("output.csv", "w", newline="") as file:
            writer = csv.DictWriter(file, fieldnames=["name", "age"])
            writer.writeheader()
            writer.writerow({"name": "Alice", "age": 25})

newline="" stops the csv module's own line endings from being doubled on Windows. More in CSV and JSON.

Working with JSON


        import json
        # write JSON
        data = {"name": "Alice", "age": 25, "city": "NYC"}
        with open("data.json", "w") as file:
            json.dump(data, file)
            # or with indentation
            json.dump(data, file, indent=2)
        # read JSON
        with open("data.json", "r") as file:
            data = json.load(file)
        # convert to/from JSON string
        json_string = json.dumps(data, indent=2)
        data = json.loads(json_string)

dump/load work with files; dumps/loads (s for string) work with strings. Use one dump per file: calling it twice, as above, writes two JSON documents back to back.

JSON: custom encoding


        # custom encoding
        import datetime
        class DateEncoder(json.JSONEncoder):
            def default(self, obj):
                if isinstance(obj, datetime.date):
                    return obj.isoformat()
                return super().default(obj)

Use it with json.dumps(data, cls=DateEncoder): dates become strings like "2024-05-01".

Remember

Always with

The file is closed for you, even after an error.

Choose the mode

r reads, w empties then writes, a adds at the end, x refuses to overwrite; add b for bytes.

One cursor

Reads and writes move it; tell() reports it and seek() moves it.

Right tool

os.path/pathlib for paths, shutil for folders, csv and json for structured data, tempfile for scratch files.
